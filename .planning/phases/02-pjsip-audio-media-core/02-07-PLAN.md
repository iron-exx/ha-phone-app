<objective>
Build the iOS UI for exactly the 5 CALL-01..05 controls (D-11): a reusable `DialpadView` (D-12/D-13/D-14), an Outgoing Call screen that properly reports outgoing calls to CallKit via `CXStartCallAction` (a gap Plan 05 didn't cover — Plan 05 only wired incoming-call actions), and an Active Call screen hosting Mute/Hold/Audio-Routing/Keypad/Transfer/End-Call. Mute and Hold route through `CXCallController` requests so CallKit's own system UI stays in sync with app state; Transfer calls `SipCallController.transfer` directly (no dedicated CallKit action exists for blind transfer initiation the way it does for hold/mute/DTMF).     import Foundation
    ```swift
    import Combine

    /// Accumulates dialed digits for the 3 reuse contexts (D-12/D-13/D-14):
    /// outgoing-call dialing, in-call DTMF entry, blind-transfer target entry.
    /// Mirrors Android's DialedNumberState.kt exactly.
    final class DialedNumberState: ObservableObject {
        @Published private(set) var current: String = ""

        func append(_ char: Character) {
            let sanitized = DialString.sanitize(String(char))
            if !sanitized.isEmpty { current += sanitized }
        }

        func backspace() {
            if !current.isEmpty { current.removeLast() }
        }

        func clear() {
            current = ""
        }

        func toCallUri(domain: String) throws -> String {
            try DialString.toSipUri(sanitized: current, domain: domain)
        }
    }
    ```

    ```swift
    /// Shared call-active flag observed by HAPhoneTestAppApp's WindowGroup to
    /// present/dismiss ActiveCallView (blocker fix, checker iteration 3 --
    /// without this, ActiveCallView exists but there is no
    /// navigation/presentation path to it on either call direction:
    /// OutgoingCallView.placeCall() only issues a CXCallController request,
    /// and CallProviderDelegate's action handlers only touched
    /// sipCallController/action.fulfill(), never SwiftUI navigation state).
    /// Deliberately a plain singleton ObservableObject, mirroring
    /// AppDelegate.shared's existing minimal-DI style in this throwaway app
    /// -- not a full DI container.
    final class CallSessionState: ObservableObject {
        static let shared = CallSessionState()
        @Published var isCallActive: Bool = false
        private init() {}
    }

    private(set) var activeCallUUID: UUID?

    func provider(_ provider: CXProvider, perform action: CXAnswerCallAction) {
        activeCallUUID = action.callUUID
        let succeeded = sipCallController.answer()
        action.fulfill()
        if succeeded {
            CallSessionState.shared.isCallActive = true
        } else {
            let endAction = CXEndCallAction(call: action.callUUID)
            CXCallController().request(CXTransaction(action: endAction)) { _ in }
        }
    }

    func provider(_ provider: CXProvider, perform action: CXEndCallAction) {
        sipCallController.hangup()
        activeCallUUID = nil
        CallSessionState.shared.isCallActive = false
        action.fulfill()
    }

    func provider(_ provider: CXProvider, perform action: CXStartCallAction) {
        activeCallUUID = action.callUUID
        provider.reportOutgoingCall(with: action.callUUID, startedConnectingAt: Date())
        do {
            try sipCallController.makeCall(action.handle.value)
            action.fulfill()
            provider.reportOutgoingCall(with: action.callUUID, connectedAt: Date())
            CallSessionState.shared.isCallActive = true
        } catch {
            action.fail()
        }
    }
    ```
